#include <iostream>
#include <set>
using namespace std;
void solve(){
	int N;
	cin>>N;
	set <int> s;
	while (N--){
		int ctr;
		cin>>ctr;
		s.insert(ctr);
	}
	cout<<s.size()<<endl;
	for (int i:s){
		cout<<i<<" ";
	}
}
int main() {
    int tc =1;
    while(tc--){
    	solve();
    }
    return 0;
}
