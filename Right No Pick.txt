import java.util.*;
class Solution{
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int c = sc.nextInt();
        int mat[][] = new int[r][c];
        long dp[][] = new long[r+1][c];
        for(int i=0; i<r; i++){
            for(int j=0; j<c; j++){
                mat[i][j] = sc.nextInt();
                if(j == 0){
                    dp[i][j] = mat[i][j];
                }
            }
        }
        for(int j=1; j<c; j++){
            long fsMax[] = getFS(dp, j-1, r);
            for(int i=0; i<r; i++){
                dp[i][j] = mat[i][j] + (dp[i][j-1] == fsMax[0]?fsMax[1]:fsMax[0]);
            }
        }
        System.out.println(getFS(dp, c-1, r)[0]);
    }
    private static long[] getFS(long [][]dp, int C, int R){
        long max = Math.max(dp[0][C], dp[1][C]);
        long smax = Math.min(dp[0][C], dp[1][C]);
        for(int i=2; i<R; i++){
            if(dp[i][C]>max){
                smax = max;
                max = dp[i][C];
            } else if(dp[i][C]>smax){
                smax = dp[i][C];
            }
        }
        return new long[]{max, smax};
    }
}
