import java.util.*;
class Solution{
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int arr[] = new int[n];
        for(int i=0; i<n; i++){
            arr[i] = sc.nextInt();
        }
        int max = Math.max(arr[0], arr[1]);
        int secMax = Math.min(arr[0], arr[1]);
        
        for(int i=2; i<n; i++){
            if(arr[i] > max){
                secMax = max;
                max = arr[i];
            } else if(arr[i] > secMax){
                secMax = arr[i];
            }
        }
        System.out.println(secMax);
    }
}
