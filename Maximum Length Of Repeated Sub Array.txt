class Solution {
    public int findLength(int[] nums1, int[] nums2) {
        int r=nums1.length;
        int c=nums1.length;
        int dp[][]=new int[r+1][c+1];
        int max=0;
        for(int i=1;i<=r;i++){
            for(int j=1;j<=c;j++){
                if(nums1[i-1]==nums2[j-1]){
                    dp[i][j]=1+dp[i-1][j-1];
                    max=Math.max(max,dp[i][j]);
                }
            }
        }
        return max;
    }
}
