#include <stdio.h>
#include <stdlib.h>
int main() {
    char str[100001];
    scanf("%s", str);
    int flag = 0;
    for(int i=0; str[i]; i++){
        char ch = str[i];
        if(ch >= 'a' && ch <= 'z'){
            flag = flag | (1<< (ch - 'a'));
        }
    }
    if(flag == ((1<<26)-1)){
        printf("YES");
    } else {
        printf("NO");
    }
    return 0;
}
