class Solution {
    public int trap(int[] height) {
        int n = height.length;
        int left[] = new int[n];
        int right[] = new int[n];
        int maxl = -1;
        int maxr = -1;
        for(int i=0; i<n; i++){
            maxl = Math.max(maxl, height[i]);
            left[i] = maxl;
        }
        for(int i=n-1; i>=0; i--){
            maxr = Math.max(maxr, height[i]);
            right[i] = maxr;
        }
        int waterStored = 0;
        for(int i=0; i<n; i++){
            int currentMin = Math.min(left[i], right[i]);
            waterStored += currentMin - height[i];
        }
        return waterStored;
    }
}
