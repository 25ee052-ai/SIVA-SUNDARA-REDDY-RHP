#include<iostream>
#include<vector>
using namespace std;
int main(){
	int N; cin >> N;
	vector<int> arr(N+1);
	for(int i=1; i<=N; i++) cin >> arr[i];
	int scnt=0, exch = 0;
	for(int i=1; i<=N; i++){
		if(arr[i] == i) scnt++;
		else if(i == arr[arr[i]]) exch++;
	}
	cout << exch/2 + 1LL*scnt*(scnt-1)/2;
}
