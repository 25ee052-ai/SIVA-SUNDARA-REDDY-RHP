#include<iostream>
#include<vector>
using namespace std;
const int diff[] = {0, 1, 0, -1, 0};
void set(vector<vector<char>>& a, int row, int col, int R, int C){
	a[row][col] = '#';
	for(int i=0; i<3; i++){
		int dr = row+diff[i], dc = col+diff[i+1];
		if(dr>=0 && dr<R && dc>=0 && dc < C && a[dr][dc] == '.'){
			set(a, dr, dc, R, C);
		}
	}
}
int main(){
	int R, C; cin >> R >> C;
	vector<vector<char>> a(R, vector<char>(C, 0));
	for(int row=0; row<R; row++){
		for(int col=0; col<C; col++){
			cin >> a[row][col];
		}
	}
	int count = 0;
	for(int row=0; row<R; row++){
		for(int col=0; col<C; col++){
			if(a[row][col] == '.'){
				count++;
				set(a, row, col, R, C);
			}
		}
	}
	cout << count << endl;
}
