#include <stdio.h>
#include <stdlib.h>
int main() {
    char str[100001];
    scanf("%s", str);
    int flag1 = 0;
    int flag2 = 0;
    for(int i=0; str[i]; i++){
        char ch = str[i];
        if(ch >= 'a' && ch <= 'z'){
            flag1 = flag1 | (1<< (ch - 'a'));
        } else {
            flag2 = flag2 | (1<< (ch - 'A'));
        }
    }
    if(flag1 == ((1<<26)-1) && flag2 == ((1<<26)-1)){
        printf("YES");
    } else {
        printf("NO");
    }
    return 0;
}
